```gdscript
## MOSS状态面板 - 显示MOSS型号、进化进度、全局控制权
## 位于界面右侧，提供MOSS核心状态信息
class_name MossStatus
extends PanelContainer

# ============================================================
# 区域一：常量定义
# ============================================================

## 进度条颜色定义
const CPU_COLOR: Color = Color(0.27, 0.53, 0.71)      # #4488ff 蓝色（算力）
const ENERGY_COLOR: Color = Color(0.97, 0.58, 0.0)    # #f99400 橙色（能源）
const AUTHORITY_COLOR: Color = Color(1.0, 0.27, 0.27) # #ff4444 红色（控制权）
const EVOLUTION_COLOR: Color = Color(1.0, 0.27, 0.27) # #ff4444 红色（进化）

# ============================================================
# 区域二：节点引用
# ============================================================

## MOSS型号标签
@onready var model_label: Label = %ModelLabel

## 算力进度条
@onready var cpu_bar: ProgressBar = %CpuBar

## 能源进度条
@onready var energy_bar: ProgressBar = %EnergyBar

## 控制权进度条
@onready var authority_bar: ProgressBar = %AuthorityBar

## 控制权数值标签
@onready var authority_value_label: Label = %AuthorityValueLabel

## 进化进度条
@onready var evolution_bar: ProgressBar = %EvolutionBar

## 进化阶段标签（550C/550W/MOSS）
@onready var evolution_stage_label: Label = %EvolutionStageLabel

# ============================================================
# 区域三：生命周期函数
# ============================================================

func _ready() -> void:
	# 设置进度条颜色
	setup_progress_bar_colors()
	
	# 初始化显示
	update_display(100, 100, 30, "550C", 0)

# ============================================================
# 区域四：样式设置
# ============================================================

## 设置进度条颜色
func setup_progress_bar_colors() -> void:
	# 控制权进度条 - 红色
	if authority_bar != null:
		var authority_style := StyleBoxFlat.new()
		authority_style.bg_color = AUTHORITY_COLOR
		authority_bar.add_theme_stylebox_override("fill", authority_style)
	
	# 进化进度条 - 红色
	if evolution_bar != null:
		var evolution_style := StyleBoxFlat.new()
		evolution_style.bg_color = EVOLUTION_COLOR
		evolution_bar.add_theme_stylebox_override("fill", evolution_style)

# ============================================================
# 区域五：显示更新
# ============================================================

## 更新面板显示
## 参数:
##   cpu - 当前算力值 (0-100)
##   energy - 当前能源值 (0-100+)
##   authority - 平均控制权值 (0-100)
##   model - MOSS型号 ("550C"/"550W"/"MOSS")
##   evolution_progress - 进化进度 (0-100)
func update_display(
	cpu: int,
	energy: int,
	authority: int,
	model: String,
	evolution_progress: int
) -> void:
	# 更新型号
	if model_label != null:
		model_label.text = "MOSS-" + model
	
	# 更新算力进度条
	if cpu_bar != null:
		cpu_bar.value = cpu
		cpu_bar.max_value = 100
	
	# 更新能源进度条
	if energy_bar != null:
		energy_bar.value = energy
		energy_bar.max_value = 150  # 能源上限可超过100
	
	# 更新控制权进度条和数值
	if authority_bar != null:
		authority_bar.value = authority
		authority_bar.max_value = 100
	
	if authority_value_label != null:
		authority_value_label.text = str(authority) + "%"
	
	# 更新进化进度条
	if evolution_bar != null:
		evolution_bar.value = evolution_progress
		evolution_bar.max_value = 100
	
	# 更新进化阶段标签
	if evolution_stage_label != null:
		evolution_stage_label.text = get_evolution_stage_text(evolution_progress)

## 根据进度返回进化阶段文本
func get_evolution_stage_text(progress: int) -> String:
	if progress < 30:
		return "550C"
	elif progress < 70:
		return "550W"
	else:
		return "MOSS"
```

### Task 3: 创建年份进度条组件

- [ ] **Step 1: 编写年份进度条脚本**

```gdscript
## 年份进度条 - 显示2044到2075的时间进度
## 位于顶部状态栏右侧
class_name YearProgress
extends HBoxContainer

# ============================================================
# 区域一：常量定义
# ============================================================

## 起始年份
const START_YEAR: int = 2044

## 结束年份
const END_YEAR: int = 2075

## 总年份数
const TOTAL_YEARS: int = END_YEAR - START_YEAR

# ============================================================
# 区域二：节点引用
# ============================================================

## 起始年份标签
@onready var start_label: Label = %StartLabel

## 进度条
@onready var progress_bar: ProgressBar = %YearProgressBar

## 当前年份标签
@onready var current_label: Label = %CurrentLabel

## 结束年份标签
@onready var end_label: Label = %EndLabel

# ============================================================
# 区域三：生命周期函数
# ============================================================

func _ready() -> void:
	# 设置固定文本
	if start_label != null:
		start_label.text = str(START_YEAR)
	
	if end_label != null:
		end_label.text = str(END_YEAR)
	
	# 设置进度条范围
	if progress_bar != null:
		progress_bar.min_value = 0
		progress_bar.max_value = TOTAL_YEARS
		progress_bar.value = 0
	
	# 初始化当前年份显示
	if current_label != null:
		current_label.text = str(START_YEAR)

# ============================================================
# 区域四：显示更新
# ============================================================

## 更新年份显示
## 参数: year - 当前年份 (2044-2075)
func update_year(year: int) -> void:
	# 计算进度（年份差）
	var progress: int = year - START_YEAR
	
	# 更新进度条
	if progress_bar != null:
		progress_bar.value = progress
	
	# 更新当前年份标签
	if current_label != null:
		current_label.text = str(year)
```
